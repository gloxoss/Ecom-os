## Architecture Laws — Violations = Failure

1. **No UI component imports an engine-specific module.** UI uses `@ecom-os/engine` only.
2. **Server Components by default.** `'use client'` only for hooks, events, browser APIs.
3. **No cross-addon imports.** i18n cannot import from multi-payment or any other addon.
4. **Graceful degradation.** If i18n fails to load a key, fall back to English — never crash.
5. **No hardcoded colors.** Tailwind design tokens only (`text-foreground`, `bg-background`).
6. **Tailwind v4 logical properties for ALL layout CSS.** No `margin-left`, `text-left`, `left:` — use `ms-`, `text-start`, `inset-inline-start`.
7. **No `any` types.** Use `// ponytail: <reason>` only when absolutely unavoidable.
8. **No `console.log` in committed code.** All logging goes through the structured logger (create `src/lib/logger.ts` with a minimal implementation here; Section 10 will extend it).
9. Every addon exports `register(ctx: AddonContext)` + `disable(): void`. CI will reject otherwise.
10. Addon errors MUST NOT crash the storefront. Wrap all addon logic in try/catch.

### Task 0 — Update `generate-addons.ts` to support packages/addons-*

**File:** `scripts/generate-addons.ts` (edit)

The generator currently only looks in `plugins/`. Extend it to also resolve addon names to `packages/addons-{name}/` so `ADDONS=i18n` finds `packages/addons-i18n/`.

```typescript
import fs from 'fs';
import path from 'path';

const addonsEnv = process.env.ADDONS || '';
const activeAddons = addonsEnv.split(',').filter(Boolean);
const pluginsDir = path.join(__dirname, '../plugins');
const packagesDir = path.join(__dirname, '../packages');
const outPath = path.join(__dirname, '../apps/storefront/src/addons.generated.ts');

function resolveAddon(name: string): { importPath: string } | null {
  // Try packages/addons-{name} first (formal packages)
  const pkgPath = path.join(packagesDir, `addons-${name}`);
  if (fs.existsSync(pkgPath)) {
    return { importPath: `../../../../packages/addons-${name}/src` };
  }
  // Fall back to plugins/{name} (legacy plugins)
  const pluginPath = path.join(pluginsDir, name);
  if (fs.existsSync(pluginPath)) {
    return { importPath: `../../../plugins/${name}` };
  }
  return null;
}

const resolvedAddons = activeAddons.map(name => {
  const resolved = resolveAddon(name);
  if (!resolved) {
    console.error(`Error: Addon '${name}' not found in packages/addons-${name}/ or plugins/${name}/`);
    process.exit(1);
  }
  return { name, ...resolved };
});

const imports = resolvedAddons
  .map((a, i) => `import * as addon${i} from '${a.importPath}';`)
  .join('\n');
const exportsArr = `export const registeredAddons = [${resolvedAddons.map((_, i) => `addon${i}`).join(', ')}];`;

fs.mkdirSync(path.dirname(outPath), { recursive: true });
fs.writeFileSync(outPath, `${imports}\n\n${exportsArr}\n`);
console.log(`Generated addons.generated.ts with ${resolvedAddons.length} addons: ${resolvedAddons.map(a => a.name).join(', ')}`);
```

### Task 3 — Update `middleware.ts` to integrate next-intl routing

**File:** `apps/storefront/src/middleware.ts` (full rewrite)

```typescript
import createMiddleware from 'next-intl/middleware';
import { NextResponse } from 'next/server';
import type { NextRequest } from 'next/server';
import { routing } from '@ecom-os/addons-i18n/src/routing';

const AUTH_COOKIE = 'ecom_auth';
const PUBLIC_ACCOUNT_PATHS = ['/account/login', '/account/register'];

const intlMiddleware = createMiddleware(routing);

export function middleware(request: NextRequest) {
  const { pathname } = request.nextUrl;
  const token = request.cookies.get(AUTH_COOKIE)?.value;

  // Strip locale prefix for auth check (/fr/account → /account)
  const pathnameWithoutLocale = pathname.replace(/^\/(fr|ar)/, '') || '/';

  const isProtectedAccount = pathnameWithoutLocale.startsWith('/account') &&
    !PUBLIC_ACCOUNT_PATHS.some(p => pathnameWithoutLocale.startsWith(p));

  if (isProtectedAccount && !token) {
    const loginUrl = request.nextUrl.clone();
    // Preserve locale prefix in redirect
    const localePrefix = pathname.match(/^\/(fr|ar)/)?.[0] ?? '';
    loginUrl.pathname = `${localePrefix}/account/login`;
    loginUrl.searchParams.set('from', pathname);
    return NextResponse.redirect(loginUrl);
  }

  return intlMiddleware(request);
}

export const config = {
  matcher: ['/((?!_next|api|favicon.ico|.*\\..*).*)'],
};
```

import { NextIntlClientProvider } from 'next-intl';
import { getMessages } from 'next-intl/server';
import { notFound } from 'next/navigation';
import { locales, isRtl } from '@ecom-os/addons-i18n/src/config';
import type { Locale } from '@ecom-os/addons-i18n/src/config';

export function generateStaticParams() {
  return locales.map(locale => ({ locale }));

export default async function LocaleLayout({
  children,
  params,
}: {
  children: React.ReactNode;
  params: Promise<{ locale: string }>;
}) {
  const { locale } = await params;
  if (!locales.includes(locale as Locale)) notFound();

  const messages = await getMessages();

  return (
    <NextIntlClientProvider messages={messages}>
      <div dir={isRtl(locale as Locale) ? 'rtl' : 'ltr'}>
        {children}
      </div>
    </NextIntlClientProvider>
  );
}

### Task 6 — Language switcher component

'use client';

import { useLocale } from 'next-intl';
import { useRouter, usePathname } from '@ecom-os/addons-i18n/src/navigation';
import { locales } from '@ecom-os/addons-i18n/src/config';

const LOCALE_LABELS: Record<string, string> = { en: 'EN', fr: 'FR', ar: 'ع' };

export function LocaleSwitcher() {
  const locale = useLocale();
  const router = useRouter();
  const pathname = usePathname();

  return (
    <div className="flex items-center gap-1">

          className={`text-xs px-2 py-1 rounded transition ${l === locale ? 'bg-foreground text-background' : 'text-secondary hover:text-foreground'}`}
          aria-label={`Switch to ${l}`}
        >
          {LOCALE_LABELS[l]}
        </button>
      ))}
    </div>
  );
}

// ponytail: minimal logger — Section 10 adds PII redaction and JSON formatting
type Level = 'info' | 'warn' | 'error';

function log(level: Level, message: string, data?: Record<string, unknown>) {
  const entry = { level, message, timestamp: new Date().toISOString(), ...data };
  if (level === 'error') console.error(JSON.stringify(entry));
  else if (level === 'warn') console.warn(JSON.stringify(entry));
  else console.log(JSON.stringify(entry));
}

export const logger = {
  info: (msg: string, data?: Record<string, unknown>) => log('info', msg, data),
  warn: (msg: string, data?: Record<string, unknown>) => log('warn', msg, data),
  error: (msg: string, data?: Record<string, unknown>) => log('error', msg, data),
};
